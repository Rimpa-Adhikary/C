#include<stdio.h>
int main()
{
	int n;
	printf("For Pattern, Enter No. of lines:");
	scanf("%d",&n);
	for(int r=0; r<=n; r++)
	{
		for(int c=0; c<=r; c++)
		{
			printf("* ");
		}
		printf("\n");
	}
return 0;
}
